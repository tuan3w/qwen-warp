def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """Merge overlapping intervals in a list of [start, end] pairs."""
    if not intervals:
        return []
    
    # Sort by start time
    intervals.sort(key=lambda x: x[0])
    
    merged = [intervals[0]]
    for current in intervals[1:]:
        last = merged[-1]
        if current[0] <= last[1]:
            # Overlapping: extend the end if needed
            last[1] = max(last[1], current[1])
        else:
            merged.append(current)
    
    return merged
```

def test_empty_list():
    assert merge_intervals([]) == []

    assert merge_intervals([[1, 3]]) == [[1, 3]]

    assert merge_intervals([[1, 2], [4, 5]]) == [[1, 2], [4, 5]]

def test_overlapping_intervals():
    assert merge_intervals([[1, 3], [2, 6], [8, 10]]) == [[1, 6], [8, 10]]

def test_contained_interval():
    assert merge_intervals([[1, 10], [2, 5]]) == [[1, 10]]

def test_adjacent_intervals():
    # Adjacent (end == start) are considered overlapping
    assert merge_intervals([[1, 4], [4, 5]]) == [[1, 5]]

def test_unsorted_input():
    assert merge_intervals([[8, 10], [1, 3], [2, 6]]) == [[1, 6], [8, 10]]
```

**How it works:** Sort intervals by start time, then iterate through them. If the current interval's start is ≤ the last merged interval's end, they overlap—extend the last interval's end if needed. Otherwise, start a new merged interval.
